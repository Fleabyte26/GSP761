# GSP761 – Build a Serverless REST API with Cloud Run and Firestore (Pet Theory)

## Known lab bugs (read first)

1. The Dockerfile line "FROM Distroless base image" isn't a real image name, so the build fails as written. The real image is `gcr.io/distroless/base-debian12`. Fixed below.
2. The second main.go has `"\"Filled in at lab startup\""` where your project ID should be, so the API can't reach Firestore. Fixed below: your real project ID is filled in automatically.
3. The lab's commands use dollar-sign variables, which get mangled when copied. Everything below uses saved values instead.
4. The region changes per lab instance. Find it in the lab text, for example the `--location` line of the repository command.
5. The second main.go is long. Big pastes get garbled in Cloud Shell, so it's split into three blocks, matching the lab's three pieces.
6. The checker lags. If a check shows 0 right after a successful step, wait 30–60 seconds and click again.

## Task 4 – Connect the REST API to Firestore

Block 1 – main.go, part 1 (imports and main):

```bash
cd ~/pet-theory/lab08
cat > main.go <<'EOF'
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"

	"cloud.google.com/go/firestore"
	"github.com/gorilla/handlers"
	"github.com/gorilla/mux"
	"google.golang.org/api/iterator"
)

var client *firestore.Client

func main() {
	var err error
	ctx := context.Background()
	client, err = firestore.NewClient(ctx, "PROJECT_X")
	if err != nil {
		log.Fatalf("Error initializing Cloud Firestore client: %v", err)
	}

	port := os.Getenv("PORT")
	if port == "" {
		port = "8080"
	}

	r := mux.NewRouter()
	r.HandleFunc("/v1/", rootHandler)
	r.HandleFunc("/v1/customer/{id}", customerHandler)

	log.Println("Pets REST API listening on port", port)
	cors := handlers.CORS(
		handlers.AllowedHeaders([]string{"X-Requested-With", "Authorization", "Origin"}),
		handlers.AllowedOrigins([]string{"https://storage.googleapis.com"}),
		handlers.AllowedMethods([]string{"GET", "HEAD", "POST", "OPTIONS", "PATCH", "CONNECT"}),
	)

	if err := http.ListenAndServe(":"+port, cors(r)); err != nil {
		log.Fatalf("Error launching Pets REST API server: %v", err)
	}
}
EOF
sed -i "s|PROJECT_X|`cat ~/project`|" main.go
grep NewClient main.go
```

The grep should show your project ID, not PROJECT_X.

Block 2 – main.go, part 2 (handlers):

```bash
cd ~/pet-theory/lab08
cat >> main.go <<'EOF'

func rootHandler(w http.ResponseWriter, r *http.Request) {
	fmt.Fprintf(w, "{status: 'running'}")
}

func customerHandler(w http.ResponseWriter, r *http.Request) {
	id := mux.Vars(r)["id"]
	ctx := context.Background()
	customer, err := getCustomer(ctx, id)
	if err != nil {
		w.WriteHeader(http.StatusInternalServerError)
		fmt.Fprintf(w, `{"status": "fail", "data": '%s'}`, err)
		return
	}
	if customer == nil {
		w.WriteHeader(http.StatusNotFound)
		msg := fmt.Sprintf("`Customer \"%s\" not found`", id)
		fmt.Fprintf(w, fmt.Sprintf(`{"status": "fail", "data": {"title": %s}}`, msg))
		return
	}
	amount, err := getAmounts(ctx, customer)
	if err != nil {
		w.WriteHeader(http.StatusInternalServerError)
		fmt.Fprintf(w, `{"status": "fail", "data": "Unable to fetch amounts: %s"}`, err)
		return
	}
	data, err := json.Marshal(amount)
	if err != nil {
		w.WriteHeader(http.StatusInternalServerError)
		fmt.Fprintf(w, `{"status": "fail", "data": "Unable to fetch amounts: %s"}`, err)
		return
	}
	fmt.Fprintf(w, fmt.Sprintf(`{"status": "success", "data": %s}`, data))
}
EOF
grep -c "func " main.go
```

Block 3 – main.go, part 3 (Customer support):

```bash
cd ~/pet-theory/lab08
cat >> main.go <<'EOF'

type Customer struct {
	Email string `firestore:"email"`
	ID    string `firestore:"id"`
	Name  string `firestore:"name"`
	Phone string `firestore:"phone"`
}

func getCustomer(ctx context.Context, id string) (*Customer, error) {
	query := client.Collection("customers").Where("id", "==", id)
	iter := query.Documents(ctx)

	var c Customer
	for {
		doc, err := iter.Next()
		if err == iterator.Done {
			break
		}
		if err != nil {
			return nil, err
		}
		err = doc.DataTo(&c)
		if err != nil {
			return nil, err
		}
	}
	return &c, nil
}

func getAmounts(ctx context.Context, c *Customer) (map[string]int64, error) {
	if c == nil {
		return map[string]int64{}, fmt.Errorf("Customer should be non-nil: %v", c)
	}
	result := map[string]int64{
		"proposed": 0,
		"approved": 0,
		"rejected": 0,
	}
	query := client.Collection(fmt.Sprintf("customers/%s/treatments", c.Email))
	if query == nil {
		return map[string]int64{}, fmt.Errorf("Query is nil: %v", c)
	}
	iter := query.Documents(ctx)
	for {
		doc, err := iter.Next()
		if err == iterator.Done {
			break
		}
		if err != nil {
			return nil, err
		}
		treatment := doc.Data()
		result[treatment["status"].(string)] += treatment["cost"].(int64)
	}
	return result, nil
}
EOF
grep -c "func " main.go
```

The count should be 5 (main, rootHandler, customerHandler, getCustomer, getAmounts).
